/** * SCRIPT PENGAMBIL LINK VIDEO GOOGLE DRIVE OTOMATIS * * Cara Pakai: * 1. Isi FOLDER_ID dengan ID folder Google Drive Anda * 2. Isi SHEET_NAME dengan nama Google Sheet tujuan * 3. Jalankan fungsi scanVideoFolder() */ // ====== KONFIGURASI - GANTI BAGIAN INI ====== const FOLDER_ID = "1Uf__AjSMlVbi-n9Qek8yde7MwBwlG22Z_BXaFwq5ckY"; // ID folder Google Drive const SHEET_NAME = "Database Video"; // Nama Google Sheet (akan dibuat otomatis jika belum ada) const SCAN_SUBFOLDER = true; // true = scan juga isi subfolder, false = hanya folder utama saja const VIDEO_EXTENSIONS = ["mp4", "mkv", "avi", "mov", "webm", "flv", "wmv", "m4v", "3gp", "ts"]; // Format video yang di-scan // ============================================ function scanVideoFolder() { const folder = DriveApp.getFolderById(FOLDER_ID); const sheet = getOrCreateSheet(SHEET_NAME); // Header sheet.clear(); sheet.getRange(1, 1, 1, 4).setValues([["Judul", "Link Video", "Kategori", "Folder"]]); sheet.getRange(1, 1, 1, 4).setFontWeight("bold").setBackground("#4285F4").setFontColor("white"); const videos = []; scanFolder(folder, folder.getName(), videos); if (videos.length === 0) { SpreadsheetApp.getUi().alert("Tidak ada video ditemukan di folder ini."); return; } // Tulis ke sheet const data = videos.map(v => [v.title, v.link, v.category, v.folder]); sheet.getRange(2, 1, data.length, 4).setValues(data); // Auto-resize kolom sheet.autoResizeColumns(1, 4); // Tampilkan hasil const sheetUrl = sheet.getParent().getUrl(); SpreadsheetApp.getUi().alert( `✅ BERHASIL!\n\n` + `Ditemukan ${videos.length} video.\n\n` + `Data sudah ditulis ke Google Sheet:\n${sheetUrl}\n\n` + `Langkah selanjutnya:\n` + `1. Buka Sheet tersebut\n` + `2. File > Share > Publish to web > pilih CSV > Publish\n` + `3. Salin link CSV dan tempel ke website Anda` ); } function scanFolder(folder, category, result) { // Scan file video di folder ini const files = folder.getFiles(); while (files.hasNext()) { const file = files.next(); const name = file.getName(); const ext = name.split('.').pop().toLowerCase(); if (VIDEO_EXTENSIONS.includes(ext)) { result.push({ title: name.replace(/\.[^/.]+$/, ""), // Hapus ekstensi dari judul link: file.getId(), // Cukup ID saja, website akan otomatis format category: category, folder: folder.getName() }); } } // Scan subfolder jika diaktifkan if (SCAN_SUBFOLDER) { const subfolders = folder.getFolders(); while (subfolders.hasNext()) { const sub = subfolders.next(); scanFolder(sub, category + " / " + sub.getName(), result); } } } function getOrCreateSheet(name) { const files = DriveApp.getFilesByName(name); let ss; if (files.hasNext()) { ss = SpreadsheetApp.open(files.next()); } else { ss = SpreadsheetApp.create(name); // Pindahkan ke folder yang sama dengan video (opsional) try { const folder = DriveApp.getFolderById(FOLDER_ID); const file = DriveApp.getFileById(ss.getId()); folder.addFile(file); DriveApp.getRootFolder().removeFile(file); } catch(e) {} } return ss.getSheets()[0]; } // Menu kustom di Google Sheet (bonus) function onOpen() { SpreadsheetApp.getUi() .createMenu('🎬 Video Tools') .addItem('Scan Video dari Drive', 'scanVideoFolder') .addToUi(); }
